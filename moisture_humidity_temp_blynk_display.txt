#define BLYNK_TEMPLATE_NAME "Soil Moisture Monitor"
#define BLYNK_AUTH_TOKEN    "HspcCSoyMLwzcodYdUhYFQxFWhcXuGlx"

#include <WiFi.h>
#include <WiFiClient.h>
#include <BlynkSimpleEsp32.h>
#include <DHT.h>
#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SH110X.h>

#include "soc/soc.h"
#include "soc/rtc_cntl_reg.h"

Adafruit_SH1106G display = Adafruit_SH1106G(128, 64, &Wire, -1);

char auth[] = BLYNK_AUTH_TOKEN;
char ssid[] = "Kawsar9";
char pass[] = "123454321";

#define SOIL_PIN 34
#define DHT_PIN 4
#define DHT_TYPE DHT22

DHT dht(DHT_PIN, DHT_TYPE);
BlynkTimer timer;

void sendSensorData() {
  // Read Soil Moisture 
  int soilValue = analogRead(SOIL_PIN);
  int soilMoisture = map(soilValue, 4095, 1500, 0, 100);
  soilMoisture = constrain(soilMoisture, 0, 100);

  // Read DHT22 
  float humidity = dht.readHumidity();
  float temperature = dht.readTemperature();

  // Validate DHT readings to avoid pushing NaN to Blynk
  if (isnan(humidity) || isnan(temperature)) {
    Serial.println("Warning: Failed to read from DHT sensor!");
  } else {
    Serial.println("================================");
    Serial.print("Soil Moisture: "); Serial.print(soilMoisture); Serial.println("%");
    Serial.print("Air Humidity: "); Serial.print(humidity); Serial.println("%");
    Serial.print("Temperature: "); Serial.print(temperature); Serial.println(" C");

    // Update Display
    display.clearDisplay();
    
    display.setTextSize(1);
    display.setTextColor(SH110X_WHITE);
    display.setCursor(28, 0); 
    display.println("SOIL MONITOR");
    display.drawFastHLine(0, 10, 128, SH110X_WHITE);

    display.setCursor(0, 16);
    display.print("Moisture : ");
    display.print(soilMoisture);
    display.println(" %");

    display.setCursor(0, 32);
    display.print("Humidity : ");
    display.print(humidity, 1);
    display.println(" %");

    display.setCursor(0, 48);
    display.print("Temp     : ");
    display.print(temperature, 1);
    display.println(" C");

    display.display();

    // Push to Blynk only when connected
    if (Blynk.connected()) {
      Blynk.virtualWrite(V0, soilMoisture);
      Blynk.virtualWrite(V1, temperature);
      Blynk.virtualWrite(V2, humidity);
    }
  }
}

void setup() {
  WRITE_PERI_REG(RTC_CNTL_BROWN_OUT_REG, 0);

  Serial.begin(115200);
  delay(1000);

  if (!display.begin(0x3C, true)) {
    Serial.println("SH1106 allocation failed");
  } else {
    display.setRotation(0); 
    display.clearDisplay();
    display.setTextColor(SH110X_WHITE);
    display.setTextSize(1);
    display.setCursor(10, 25);
    display.println("System Starting...");
    display.display();
  }

  dht.begin();

  // Start WiFi and Blynk non-blocking connection setup
  WiFi.begin(ssid, pass);
  Blynk.config(auth);
  Blynk.connect(10000); 

  // Refresh rate to 2500ms (2.5 seconds) for stable DHT22 sampling
  timer.setInterval(2500L, sendSensorData);
}

void loop() {
  // Reconnect to Blynk automatically if Wi-Fi dropped
  if (WiFi.status() == WL_CONNECTED) {
    if (!Blynk.connected()) {
      Blynk.connect(1000);
    } else {
      Blynk.run();
    }
  }
  
  timer.run();
}
